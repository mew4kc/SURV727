As you might have seen earlier, it’s not always clear which number of clusters is the optimal choice. To automate K Means clustering, program a function based on kmeans() that takes K as an argument. You can fix the other arguments, e.g. such that a specific dataset is always used when calling the function.

automate_kmeans <- function(K, data, columns_to_cluster = NULL, visualize = TRUE) {
  # Check for valid inputs
  if (!is.numeric(K) || K <= 0) {
    stop("K must be a positive numeric value.")
  }
  
  # Select numeric columns based on the provided `columns_to_cluster` or use all numeric columns
  if (is.null(columns_to_cluster)) {
    # Default: Use all numeric columns
    data_to_cluster <- data %>%
      select(where(is.numeric)) %>%
      na.omit()
  } else {
    # Use the specified columns
    data_to_cluster <- data %>%
      select(all_of(columns_to_cluster)) %>%
      select(where(is.numeric)) %>%
      na.omit()
  }
  
  # Check if `data_to_cluster` is valid
  if (ncol(data_to_cluster) == 0) {
    stop("No valid numeric columns found for clustering.")
  }
  
  # Perform k-means clustering
  kmeans_result <- kmeans(as.matrix(data_to_cluster), centers = K, nstart = 25)
  
  # Add the cluster assignments back to the original dataset
  clustered_data <- data %>%
    mutate(cluster = NA) # Initialize cluster column
  clustered_data[rownames(data_to_cluster), "cluster"] <- kmeans_result$cluster
  
  # Visualize the clusters (optional)
  if (visualize) {
    # Visualize the optimal number of clusters (WSS method)
    print(fviz_nbclust(data_to_cluster, kmeans, method = "wss") + 
            ggtitle("Optimal Number of Clusters - WSS Method"))
    
    # Visualize the clusters
    print(fviz_cluster(kmeans_result, data = data_to_cluster) + 
            ggtitle(paste("K-Means Clustering with K =", K)))
  }
  
  # Return the dataset with cluster assignments and the kmeans result object
  return(list(clustered_data = clustered_data, kmeans_result = kmeans_result))
}
